greet("홍길동", "반갑다")  # 반갑다, 홍길동!

가변 인자 (*args, **kwargs)

# 위치 인자 여러 개
def total(*args):
    """입력된 모든 수 더하기"""
    result = 0
    for num in args:
        result += num
    return result

print(total(1, 2, 3))      # 6
print(total(1, 2, 3, 4, 5)) # 15

# 키워드 인자 여러 개
def profile(**kwargs):
    for key, value in kwargs.items():
        print(f"{key}: {value}")

profile(name="김철수", age=20, city="부산")

07. 정규표현식

re 모듈 기본 사용

import re

# 컴파일 (재사용 시 효율적)
pattern = re.compile(r'정규식패턴')

# 한 번만 쓸 때
result = re.search(r'패턴', '텍스트')

# match: 처음부터 매치
m = re.match(r'\d+', '123abc')
if m:
    print(m.group())  # 123

# search: 전체에서 찾아냄
s = re.search(r'\d+', 'abc123def')
print(s.group())  # 123

# findall: 모두 찾아서 리스트 반환
all_nums = re.findall(r'\d+', 'a1b22c333')
print(all_nums)  # ['1', '22', '333']

메타문자 완전 가이드

패턴의미매치 예시
.임의의 한 문자a.c → abc, a1c, a$c
^시작^Hello → Hello로 시작
$끝world$ → world로 끝
*0번 이상a*b → b, ab, aab, aaab
+1번 이상a+b → ab, aab (b 단독 X)
?0~1번colou?r → color, colour
{n,m}n~m번a{2,4} → aa, aaa, aaaa
[]문자셋[가-힣] 한글 하나
[^]부정[^0-9] 숫자 아님
()그룹(a|b)c → ac 또는 bc
|ORCat|Dog → Cat 또는 Dog

08. 클래스

기본 구조

class 자동차:
    # 클래스 변수 (공유됨)
    제조사 = "현대"

    def __init__(self, 색상, 속도=0):
        # 인스턴스 변수 (객체별로 다름)
        self.색상 = 색상
        self.속도 = 속도

    def 가속(self, amount):
        self.속도 += amount

    def 멈추다(self):
        self.속도 = 0

# 객체 생성
my_car = 자동차("빨강")
my_car.가속(30)
print(my_car.속도)  # 30

메서드 종류

class 예시:
    클래스변수 = 0

    def __init__(self):
        self.인스턴스변수 = 0

    # 인스턴스 메서드 (self 사용, 객체 필요)
    def 인스턴스메서드(self):
        self.인스턴스변수 += 1

    # 정적 메서드 (self 없이 사용, 객체 불필요)
    @staticmethod
    def 정적메서드():
        print("객체 없이 호출 가능")

    # 클래스 메서드 (cls 사용, 클래스 변수 접근)
    @classmethod
    def 클래스메서드(cls):
        cls.클래스변수 += 1

# 호출 예시
obj = 예시()
obj.인스턴스메서드()
예시.정적메서드()
예시.클래스메서드()

09. 예외처리

기본 구조

try:
    # 실행할 코드
    result = 10 / 0
except ZeroDivisionError:
    # 에러 처리
    print("0으로 나눌 수 없습니다")
except Exception as e:
    # 모든 에러 잡기
    print(f"에러 발생: {e}")
else:
    # 에러 없을 때 실행
    print("성공!")
finally:
    # 무조건 실행 (정리용)
    print("끝났습니다")

예외발생 상황

FileNotFoundError파일 없음

문제 5. 그리디 - 1이 될 때까지

N, K = map(int, input().split())
count = 0

while N > 1:
    if N % K == 0:
        N //= K
    else:
        N -= N % K
    count += 1

print(count)
