가변 인자 (*args)

def total(*args):
    """입력된 모든 수 더하기"""
    result = 0
    for num in args:
        result += num
    return result

print(total(1, 2, 3))      # 6
print(total(1, 2, 3, 4, 5)) # 15

# 딕셔너리 형태
def profile(**kwargs):
    for key, value in kwargs.items():
        print(f'{key}: {value}')

profile(name='홍길동', age=20)

람다 함수

    return a - b

# 리스트 정렬에 유용

students.sort(key=lambda x: x[1])  # 점수 기준 정렬
print(students)  # [('김', 85), ('홍', 90), ('이', 95)]

패턴의미매치 결과활용 예시
.한 문자a.c → abc[가-힣] 한글
*0회 이상a*b → b, ab, aab\w* 단어 0개 이상
+1회 이상a+b → ab, aab[0-9]+ 숫자 1개 이상
?0~1회colou?r → color|colourhttps? HTTP|HTTPS
[]문자셋[abc] a|b|c[가-힣]+ 한글 문자열
[^]부정[^0-9] 숫자 아님[^A-Z] 대문자 아님

\d숫자[0-9]\d{3}-\d{4}

08. 클래스

기본 구조

class 자동차:
    # 클래스 변수 (모든 객체가 공유)
    제조사 = "현대"

    def __init__(self, 색상):
        # 인스턴스 변수 (객체별로 다름)
        self.색상 = 색상
        self.속도 = 0

    def 가속(self, amount):
        self.속도 += amount

    def 멈추다(self):
        self.속도 = 0

    @staticmethod
    def 경고():
        print("경고음!")

    @classmethod
    def 정보(cls):
        print(f"제조사: {cls.제조사}")

# 객체 생성
my_car = 자동차("빨강")
my_car.가속(30)
print(my_car.속도)  # 30

09. 예외처리

try-except 구조

try:
    # 실행 코드
    result = 10 / 0
except ZeroDivisionError as e:
    # 에러 처리
    print(f"에러 발생: {e}")
except Exception as e:
    # 모든 에러 잡기
    print(f"예상치 못한 에러: {e}")
else:
    # 에러 없을 때 실행
    print("성공!")
finally:
    # 무조건 실행 (정리용)
    print("끝")

알고리즘 - 그리디

1이 될 때까지 최소 횟수

N, K = map(int, input().split())
count = 0

while N > 1:
    if N % K == 0:
        N //= K  # 나누기
    else:
        N -= N % K  # 나머지만큼 빼기
    count += 1

print(count)
