⚠️ 예외처리 완전 정리

예외(Exception)란?

프로그램 실행 중에 발생하는 오류로, 정상적인 프로그램 흐름을 방해(interrupt)하는 사건입니다.

적절한 예외처리가 없으면 프로그램이 비정상적으로 종료됩니다.

주요 예외 종류

예외 클래스발생 상황예시 코드
ZeroDivisionError0으로 나누려고 할 때10 / 0
IndexError리스트/문자열 인덱스 범위 초과[1,2,3][3]
KeyError딕셔너리에 없는 키 접근{'a':1}['b']
ValueError잘못된 값으로 함수 호출int('abc')
TypeError타입 불일치로 연산/function 호출'hello' + 5
FileNotFoundError존재하지 않는 파일 열기 시도open('없는파일.txt')
ImportError모듈을 찾을 수 없을 때import 없는모듈
KeyboardInterrupt사용자가 Ctrl+C를 눌렀을 때무한 루프 중 인터럽트

예외처리 기본 문법

기본 형태

try:
    # 예외가 발생할 수 있는 코드
    위험한_코드
except 예외타입:
    # 특정 예외가 발생했을 때 처리
    처리할_코드
else:
    # 예외가 발생하지 않았을 때 실행
    정상처리_코드
finally:
    # 예외 발생 여부와 관계없이 항상 실행
    정리_코드

다중 예외 처리

try:
    risk = 10 / int(input("숫자 입력: "))
except ZeroDivisionError:
    print("0으로 나눌 수 없습니다!")
except ValueError:
    print("올바른 숫자를 입력해주세요!")
except Exception as e:  # 모든 기타 예외 catch
    print(f"예상치 못한 오류: {e}")
else:
    print(f"결과: {risk}")
finally:
    print("프로그램 종료 정리 중...")  # 항상 실행

def get_positive_number():
    while True:
        try:
            num = float(input("양수를 입력하세요: "))

의도적인 예외 발생 (raise)

def validate_age(age):
    """나이 유효성 검사"""
    if not isinstance(age, int):
        raise TypeError("나이는 정수여야 합니다")
    if age < 0:
        raise ValueError("나이는 음수가 될 수 없습니다")
    if age > 150:
        raise ValueError("나이가 너무 많습니다 (150세를 초과)")
    return True

# 사용
try:
    validate_age(-5)
except ValueError as e:
    print(f"입력 오류: {e}")
except TypeError as e:
    print(f"타입 오류: {e}")

사용자 정의 예외 클래스

# 기본 사용자 정의 예외
class 맞춤예외(Exception):
    """기본 사용자 정의 예외 클래스"""
    pass

# 특정 정보를 포함하는 사용자 정의 예외
class ValidationError(Exception):
    """입력 검증 관련 예외"""
    def __init__(self, message, field=None):
        self.message = message
        self.field = field
        super().__init__(self.message)

class AgeValidationError(ValidationError):
    """나이 검증 관련 예외"""
    def __init__(self, message, age=None):
        self.age = age
        super().__init__(message, field="age")

# 사용
def register_user(age, username):
    if not isinstance(age, int):
        raise TypeError("나이는 정수여야 합니다")
    if age < 0:
        raise AgeValidationError("나이는 음수가 될 수 없습니다", age=age)
    if age > 150:
        raise AgeValidationError("나이가 150세를 초과할 수 없습니다", age=age)
    if not username or len(username.strip()) < 2:
        raise ValidationError("사용자 이름은 2자 이상이어야 합니다", field="username")

    return {"age": age, "username": username.strip()}

# 예외 처리
try:
    user = register_user(-5, "ab")
except AgeValidationError as e:
    print(f"나이 오류: {e.message} (입력값: {e.age})")
except ValidationError as e:
    print(f"검증 오류: {e.message} (필드: {e.field})")
except TypeError as e:
    print(f"타입 오류: {e}")

finally의 중요성

리소스 정리에 finally가 필수적입니다:

# 파일 처리 예시
file = None
try:
    file = open('data.txt', 'r', encoding='utf8')
    data = file.read()
    # 데이터 처리...
    result = process(data)
finally:
    if file is not None:
        file.close()  # 예외 발생 여부와 관계없이 파일을 닫음
        print("파일이 정상적으로 닫혔습니다")

# 또는 with 문 사용 (자동으로 finally 처리)
try:
    with open('data.txt', 'r', encoding='utf8') as f:
        data = f.read()
        result = process(data)
except FileNotFoundError:
    print("파일을 찾을 수 없습니다")
# with 문을 벗어나면 자동으로 f.close() 호출됨

연습문제

안전한 계산기 
def safe_divide(a, b):
    """0으로 나누지 않는 안전한 나눗셈"""
    try:
        return a / b
    except ZeroDivisionError:
        print("오류: 0으로 나눌 수 없습니다")
        return None
    except TypeError as e:
        print(f"오류: 숫자가 아닌 값을 입력했습니다 - {e}")
        return None

# 테스트
print(safe_divide(10, 2))   # 5.0
print(safe_divide(10, 0))   # None (오류 메시지 출력)
print(safe_divide(10, "a")) # None (오류 메시지 출력)

🎯 변형문제 - 안전한 리스트 항목 접근기

리스트와 인덱스를 받아서, 범위를 벗어나거나 잘못된 타입의 인덱스가 들어와도 프로그램이 멈추지 않고 None을 반환하며 오류 메시지를 출력하는 safe_get() 함수를 작성해보세요.

정답 보기 
def safe_get(items, index):
    """범위를 벗어나도 안전한 리스트 접근"""
    try:
        return items[index]
    except IndexError:
        print(f"오류: 인덱스 {index}는 범위를 벗어났습니다")
        return None
    except TypeError as e:
        print(f"오류: 인덱스는 정수여야 합니다 - {e}")
        return None

fruits = ['사과', '바나나', '포도']
print(safe_get(fruits, 1))     # 바나나
print(safe_get(fruits, 10))    # None (오류 메시지 출력)
print(safe_get(fruits, '1'))   # None (오류 메시지 출력)
파일 처리 프로그램 
def read_file_safely(filename):
    """파일을 안전하게 읽는 함수"""
    try:
        with open(filename, 'r', encoding='utf8') as f:
            content = f.read()
        return content, None  # (내용, 오류)
    except FileNotFoundError:
        return None, f"파일 '{filename}'을 찾을 수 없습니다"
    except PermissionError:
        return None, f"파일 '{filename}'에 대한 읽기 권한이 없습니다"
    except UnicodeDecodeError:
        return None, f"파일 '{filename}'의 인코딩을 읽을 수 없습니다"
    except Exception as e:
        return None, f"예상치 못한 오류: {e}"

# 사용
content, error = read_file_safely("data.txt")
if error:
    print(f"오류 발생: {error}")
else:
    print(f"파일 내용 (처음 100자): {content[:100]}...")

🎯 변형문제 - 안전한 API 요청 함수

웹 API에 요청을 보낼 때 발생할 수 있는 여러 오류(연결 실패, 타임아웃, 그 외 오류)를 각각 구분해서 처리하는 safe_request() 함수를 작성해보세요.

정답 보기 
import requests

def safe_request(url):
    """API 요청을 안전하게 처리하는 함수"""
    try:
        response = requests.get(url, timeout=5)
        return response.json(), None
    except requests.exceptions.ConnectionError:
        return None, "서버에 연결할 수 없습니다"
    except requests.exceptions.Timeout:
        return None, "요청 시간이 초과되었습니다"
    except requests.exceptions.RequestException as e:
        return None, f"요청 중 오류 발생: {e}"

data, error = safe_request("https://api.example.com/data")
if error:
    print(f"오류 발생: {error}")
else:
    print(f"응답 데이터: {data}")
