*args (가변 위치 인자)

def total(*args):
    """입력된 모든 수 더하기"""
    result = 0
    for num in args:
        result += num
    return result

print(total(1, 2, 3))      # 6
print(total(1, 2, 3, 4, 5)) # 15

**kwargs (가변 키워드 인자)

def profile(**kwargs):
    """키-값 쌍 출력"""
    for key, value in kwargs.items():
        print(f'{key}: {value}')

profile(name='홍길동', age=20, city='부산')
# name: 홍길동
# age: 20
# city: 부산

# 일반 함수
def minus(a, b):
    return a - b

print(minus(10, 3))  # 7

# 리스트 정렬에 유용
students = [('홍', 90), ('김', 85), ('이', 95)]

print(students)  # [('김', 85), ('홍', 90), ('이', 95)]

연습문제

거리 계산 함수 
def distance(start, end):
    """시작점과 끝점 거리 계산"""
    dx = abs(end[0] - start[0])
    dy = abs(end[1] - start[1])
    return dx, dy

start = (1, 1)
end = (5, 3)
dx, dy = distance(start, end)
print(f'dx: {dx}, dy: {dy}')  # dx: 4, dy: 2

🎯 변형문제 - 직사각형 둘레와 넓이 계산 함수

가로(width)와 세로(height)를 매개변수로 받아서, 둘레와 넓이를 동시에 반환하는 함수 rectangle_info()를 작성해보세요. (둘레 = (가로+세로)×2, 넓이 = 가로×세로)

정답 보기 
def rectangle_info(width, height):
    """직사각형의 둘레와 넓이 계산"""
    perimeter = (width + height) * 2
    area = width * height
    return perimeter, area

p, a = rectangle_info(6, 4)
print(f'둘레: {p}, 넓이: {a}')  # 둘레: 20, 넓이: 24
