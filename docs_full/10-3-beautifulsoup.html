🍲 BeautifulSoup 완전 정리

BeautifulSoup란?

BeautifulSoup은 HTML과 XML 파싱을 쉽게 만들어주는 파이썬 라이브러리입니다.

복잡한 HTML 구조에서도 원하는 데이터를 쉽게 추출할 수 있도록 도와줍니다.

설치 방법

# 터미널에서 실행
pip install beautifulsoup4

# lxml 파서 (더 빠름, 권장)
pip install lxml

기본 사용법

from bs4 import BeautifulSoup as bs
import requests

# HTML 문자열로부터 파싱
html_string = '''
<html>
<body>
    <h1>제목</h1>
    <p class="intro">소개 문단</p>
    <ul>
        <li>항목 1</li>
        <li>항목 2</li>
    </ul>
</body>
</html>
'''

soup = bs(html_string, 'html.parser')

# 요소 찾기
title = soup.find('h1')
print(title.text)  # 제목

# 또는
soup = bs('<h1>Hello</h1>', 'html.parser')
print(soup.find('h1').text)  # Hello

파서 종류

파서장점단점설치 필요 여부
html.parser기본 내장, 별도 설치 불필요상대적으로 느림아니오
lxml매우 빠르고 유연함별도 설치 필요예 (pip install lxml)
html5lib웹 표준 준수, 관대한 파싱가장 느림예 (pip install html5lib)
# lxml 파서 사용 (권장)
soup = bs(html_string, 'lxml')

# html5lib 파서 사용
soup = bs(html_string, 'html5lib')

요소 선택 방법

1. find() / find_all() - 메서드 방식

find(): 첫 번째 일치하는 요소만 반환

soup = bs('<p>첫 번째</p><p>두 번째</p>', 'html.parser')

# 첫 번째 p 태그 찾기
first_p = soup.find('p')
print(first_p.text)  # 첫 번째

# 속성을 기준으로 찾기
soup = bs('<p class="intro">소개</p><p class="content">내용</p>', 'html.parser')
intro_p = soup.find('p', class_='intro')
print(intro_p.text)  # 소개

# 여러 속성 조합
element = soup.find('div', {'class': 'container', 'id': 'main'})

find_all(): 모든 일치하는 요소들을 리스트로 반환

soup = bs('''
<p>첫 번째 단락</p>
<div class="section">
    <p>섹션 내 단락</p>
</div>
<p>두 번째 단락</p>
''', 'html.parser')

# 모든 p 태그 찾기
all_p = soup.find_all('p')
print(len(all_p))  # 3
for p in all_p:
    print(p.text)

# 특정 클래스를 가진 요소 찾기
section_divs = soup.find_all('div', class_='section')
print(len(section_divs))  # 1

# 속성이 존재하는 요소 찾기
links_with_href = soup.find_all('a', href=True)
# href 속성이 있는 모든 a 태그 찾기

# 정규표현식으로 속성 매칭
import re
pattern_links = soup.find_all('a', href=re.compile(r'https?://'))
# http 또는 https로 시작하는 링크 찾기

2. select() / select_one() - CSS 선택자 방식

CSS 선택자를 사용해 더 직관적이고 강력한 요소를 선택할 수 있습니다.

select_one(): 첫 번째 일치하는 요소만 반환

soup = bs('''

    <h1>제목</h1>

first_li = soup.select_one('li')
print(first_li.text)  # 메뉴 1

# 자손 선택자 (공백으로 구분)
menu_items = soup.select('.menu li')

    print(item.text)

# 직접 자식 선택자 (>)
direct_children = soup.select('.container > *')
# container의 직접 자식 요소들만 선택

# 인접 형제 선택자 (+)
# h1 + p : h1 바로 다음에 오는 p 요소

# 일반 형제 선택자 (~)
# h1 ~ p : h1 이후에 오는 모든 p 요소

select(): 모든 일치하는 요소들을 리스트로 반환

# 같은 타입 요소 중에서 n번째

텍스트와 속성 추출

텍스트 추출 방법

element = soup.find('h1')

# 방법 1: .text 속성
text1 = element.text
# 모든 자식 요소의 텍스트를 결합 (HTML 태그 제거)

# 방법 2: .get_text() 메서드
text2 = element.get_text()
# .text와 동일하지만 추가 옵션 제공

# 방법 3: .string 속성 (단일 텍스트 노드만)
text3 = element.string
# 요소가 오직 하나의 텍스트 노드만 가지고 있을 때 사용
# 자식 요소가 있으면 None 반환

# 텍스트 정리 옵션
clean_text = element.get_text(strip=True)  # 앞뒤 공백 제거
clean_text = element.get_text(separator=' ', strip=True)  # 공백으로 연결하고 정리

속성(Attributes) 추출 방법

element = soup.find('a', href='https://example.com')

# 방법 1: 딕셔너리처럼 접근 (가장 일반적)
href_value = element['href']
print(href_value)  # https://example.com

# 방법 2: .attrs 속성 사용
attrs_dict = element.attrs
href_value = attrs_dict['href']

# 방법 3: .get() 메서드 (안전한 접근)
href_value = element.get('href')
# 속성이 없을 때 None 반환 (KeyError 방지)
href_value = element.get('href', '기본값')  # 기본값 지정 가능

# 모든 속성 가져오기
all_attrs = element.attrs
print(all_attrs)  # {'href': 'https://example.com', 'class': ['link']}

# 특정 속성 목록 가져오기
class_list = element.get_attribute_list('class')
print(class_list)  # ['link'] (리스트 형태로 반환)

실전 예제: 뉴스 기사 크롤링

import requests
from bs4 import BeautifulSoup as bs

# 뉴스 기사 페이지 가져오기
url = "https://news.example.com/article/12345"
response = requests.get(url, headers={'User-Agent': 'Mozilla/5.0'})
response.encoding = 'utf-8'  # 한글 깨짐 방지

# HTML 파싱
soup = bs(response.text, 'html.parser')

# 기사 제목 추출
title_element = soup.find('h1', class_='article-title')
if title_element:
    title = title_element.text.strip()
    print(f"제목: {title}")

# 기사 내용 추출
content_div = soup.find('div', class_='article-content')
if content_div:
    # 모든 단락 텍스트 결합
    paragraphs = content_div.find_all('p')
    content = '\n'.join([p.text.strip() for p in paragraphs])
    print(f"내용:\n{content}")

# 작성자 및 날짜 추출
meta_div = soup.find('div', class_='article-meta')
if meta_div:
    author = meta_div.find('span', class_='author')
    date = meta_div.find('span', class_='date')
    if author:
        print(f"작성자: {author.text.strip()}")
    if date:
        print(f"날짜: {date.text.strip()}")

# 태그 목록 추출
tags_div = soup.find('div', class_='article-tags')
if tags_div:
    tags = tags_div.find_all('a', class_='tag')
    tag_list = [tag.text.strip() for tag in tags]
    print(f"태그: {', '.join(tag_list)}")

실전 예제: 상품 정보 크롤링

import requests
from bs4 import BeautifulSoup as bs

# 상품 목록 페이지
url = "https://shop.example.com/category/electronics"
response = requests.get(url)
soup = bs(response.text, 'html.parser')

# 모든 상품 카드 찾기
product_cards = soup.find_all('div', class_='product-card')
print(f"총 {len(product_cards)}개 상품 발견")

products = []
for card in product_cards:
    # 상품명
    name_element = card.find('h3', class_='product-name')
    name = name_element.text.strip() if name_element else "정보 없음"

    # 가격
    price_element = card.find('span', class_='price')
    price = price_element.text.strip() if price_element else "가격 정보 없음"

    # 이미지 URL
    img_element = card.find('img')
    img_url = img_element['src'] if img_element and img_element.has_attr('src') else "이미지 없음"

    # 평점
    rating_element = card.find('div', class_='rating')
    rating = rating_element.text.strip() if rating_element else "평가 없음"

    products.append({
        'name': name,
        'price': price,
        'image_url': img_url,
        'rating': rating
    })

# 결과 출력
for i, product in enumerate(products[:5], 1):  # 처음 5개만 출력
    print(f"{i}. {product['name']}")
    print(f"   가격: {product['price']}")
    print(f"   평점: {product['rating']}")
    print(f"   이미지: {product['image_url']}")
    print()

next_siblings = element.next_siblings  # 이후 모든 형제들
previous_siblings = element.previous_siblings  # 이전 모든 형제들

# 특정 조건의 요소 찾기

elements_with_text = soup.find_all(text=lambda text: text and 'important' in text)

    # 포스터 URL

    print(f"제목: {title}")
    print(f"연도: {year} | 장르: {genre} | 평점: {rating}")

    print("-" * 50)

import requests
from bs4 import BeautifulSoup as bs

# 주식 목록 페이지 시뮬레이션
html_content = '''
<table class="stock-table">
    <thead>
        <tr>
            <th>종목명</th>
            <th>현재가</th>
            <th>등락률</th>
            <th>거래량</th>
        </tr>
    </thead>
    <tbody>
        <tr>
            <td class="stock-name">삼성전자</td>
            <td class="price">78,500</td>
            <td class="change">+1.2%</td>
            <td class="volume">12,345,678</td>

        <tr>
            <td class="stock-name">LG전자</td>
            <td class="price">65,200</td>
            <td class="change">-0.5%</td>
            <td class="volume">5,678,901</td>
