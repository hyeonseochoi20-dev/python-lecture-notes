<작음3 < 4True

<=작거나 같음5 <= 5True
>=크거나 같음5 >= 6False

XYX or Y
TrueTrueTrue
TrueFalseTrue
FalseTrueTrue

Xnot X
TrueFalse
FalseTrue

조건문 - if

if만 사용

if 조건:
    실행할 코드

# 예시
if score >= 60:
    print('합격')

if-else 사용

if 조건:
    실행1
else:
    실행2

# 예시
score = 55
if score >= 60:
    print('합격')
else:
    print('불합격')

if-elif-else 사용

if 조건1:
    실행1
elif 조건2:
    실행2
else:
    실행3

# 예시
score = 85
if score >= 90:
    grade = 'A'
elif score >= 80:
    grade = 'B'
elif score >= 70:
    grade = 'C'
else:
    grade = 'F'
print(grade)  # B

논리 연산자 활용 예시

# AND 활용
has_pc = True
has_phone = False
if has_pc or has_phone:
    print('인터넷 검색 가능')

# OR 활용
age = 20
has_license = True
if age >= 20 and has_license:
    print('운전 가능')

# NOT 활용
is_weekend = False
if not is_weekend:
    print('평일입니다')

반복문 - while

기본 구조

# 조건이 참인 동안 반복
while 조건:
    실행할 코드

# 예시
count = 0
while count < 5:
    print(count)
    count += 1  # 중요: 무한 루프 방지!
# 출력: 0 1 2 3 4

무한 루프와 break

while True:
    user = input('명령어 입력 (quit로 종료): ')
    if user == 'quit':
        break
    print(f'입력: {user}')

반복문 - for

range 함수

# range(stop) : 0부터 stop-1까지
for i in range(5):
    print(i, end=' ')  # 0 1 2 3 4

# range(start, stop)
for i in range(2, 7):
    print(i, end=' ')  # 2 3 4 5 6

# range(start, stop, step)
for i in range(1, 11, 2):
    print(i, end=' ')  # 1 3 5 7 9

리스트 순회

fruits = ['사과', '바나나', '오렌지']

# 값만 출력
for fruit in fruits:
    print(fruit)

# 인덱스와 값 함께
for idx, fruit in enumerate(fruits, start=1):
    print(f'{idx}: {fruit}')
# 1: 사과
# 2: 바나나
# 3: 오렌지

여러 리스트 동시 순회

names = ['홍', '김', '이']
scores = [90, 85, 95]

for name, score in zip(names, scores):
    print(f'{name}: {score}')

🎯 변형문제 2 - 3의 배수이면서 5의 배수인지 판별

숫자 하나를 입력받아 3의 배수이면서 동시에 5의 배수이면 "15의 배수"를, 3의 배수만이면 "3의 배수"를, 5의 배수만이면 "5의 배수"를, 둘 다 아니면 "해당없음"을 출력하세요.

정답 보기 
num = int(input('숫자 입력: '))

if num % 3 == 0 and num % 5 == 0:
    print('15의 배수')
elif num % 3 == 0:
    print('3의 배수')
elif num % 5 == 0:
    print('5의 배수')
else:
    print('해당없음')

🎯 변형문제 3 - 장바구니에 품절 상품이 있는지 확인

장바구니 리스트와 품절 상품 리스트가 각각 주어질 때, in 연산자를 사용해서 장바구니 안에 품절된 상품이 하나라도 있는지 확인하는 코드를 작성해보세요.

정답 보기 
cart = ['우유', '계란', '두부', '식빵']
sold_out = ['식빵', '라면']

if '식빵' in cart and '식빵' in sold_out:
    print('식빵은 품절 상품입니다')

# 응용: 장바구니 중 품절 상품만 추려내기
sold_out_in_cart = [item for item in cart if item in sold_out]
print(sold_out_in_cart)  # ['식빵']
