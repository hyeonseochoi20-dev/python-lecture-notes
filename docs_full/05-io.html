📥 입력/출력 완전 정리

사용자 입력 - input()

# 기본 입력 (항상 문자열 반환)
text = input()
print(text)
print(type(text))  # <class 'str'>

# 안내문구와 함께
name = input('이름을 입력하세요: ')
print(f'안녕하세요, {name}님!')

# 숫자 입력받기 (문자열을 숫자로 변환)
age = int(input('나이: '))
height = float(input('키: '))

여러 값 한 번에 입력받기

# 공백으로 구분된 입력
a, b, c = input('세 값 입력: ').split()
# 입력: 10 20 30
# a='10', b='20', c='30'

# 숫자로 변환하면서 받기
x, y = map(int, input('두 수: ').split())
# 입력: 10 20
# x=10, y=20

# 리스트로 받기
numbers = list(map(int, input('숫자들: ').split()))
print(sum(numbers))

출력 - print()

# 기본 출력
print('Hello')
print(123)
print([1, 2, 3])

# 여러 값 출력
print('Life', 'is', 'too', 'short')
# Life is too short

# 구분자 변경
print('Life', 'is', 'too', 'short', sep='-')
# Life-is-too-short

# 줄바꿈 없이 출력
print('Life', end=' ')
print('is', end=' ')
print('too', end=' ')
print('short')

파일 입출력

파일 모드

모드설명
r읽기 (기본, 없으면 오류)
w쓰기 (없으면 생성, 있으면 덮어씀)
a추가 (기존 내용 유지)
t텍스트 모드 (기본)
b바이너리 모드

파일 읽기

# 전체 읽기
with open('file.txt', 'rt', encoding='utf8') as f:
    content = f.read()

# 한 줄씩 읽기
with open('file.txt', 'rt', encoding='utf8') as f:
    line = f.readline()

# 모든 줄 리스트로
with open('file.txt', 'rt', encoding='utf8') as f:
    lines = f.readlines()

# 한 줄씩 순회 (추천)
with open('file.txt', 'rt', encoding='utf8') as f:
    for line in f:
        print(line.strip())

파일 쓰기

# 덮어쓰기
with open('output.txt', 'wt', encoding='utf8') as f:
    f.write('새로운 내용\n')

# 추가쓰기
with open('log.txt', 'at', encoding='utf8') as f:
    f.write('새 로그\n')

# 여러 줄 쓰기
lines = ['첫째\n', '둘째\n', '셋째\n']
with open('output.txt', 'wt') as f:
    f.writelines(lines)

with 문 - 자동 close

# with를 사용하면 자동으로 close됨
with open('file.txt', 'rt') as f:
    data = f.read()
# 여기서 자동으로 f.close() 호출됨

# 인코딩 지정 (한글 파일)
with open('소나기.txt', 'rt', encoding='utf8') as f:
    text = f.read()

# ms949/cp949 (구 한글 인코딩)
with open('파일.txt', 'rt', encoding='ms949') as f:
    text = f.read()

연습문제

시스템 로그 파일 생성 
import datetime

current_time = datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S")

log_messages = [
    "사용자가 관리자 페이지에 로그인했습니다.",
    "데이터베이스 연결에 성공했습니다.",
    "경고: 디스크 용량이 80% 이상 채워졌습니다.",
    "에러: 파일 업로드 중 예기치 못한 오류가 발생했습니다."
]

print("로그 생성을 시작합니다...")

with open('system.log', 'a', encoding='utf8') as f:
    for msg in log_messages:
        f.write(f"[{current_time}] {msg}\n")

print("로그 파일 작성 완료!")

🎯 변형문제 1 - 쇼핑몰 주문 접수 기록 남기기

쇼핑몰에 주문이 들어올 때마다 order.log 파일에 "[시각] 주문번호 ORD-1001 접수" 같은 형식으로 누적 기록하는 코드를 작성해보세요. (기존 로그는 지우지 않고 이어서 씁니다)

정답 보기 
import datetime

now = datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S")
orders = ['ORD-1001', 'ORD-1002', 'ORD-1003']

with open('order.log', 'a', encoding='utf8') as f:
    for order_id in orders:
        f.write(f"[{now}] 주문번호 {order_id} 접수\n")

print("주문 접수 기록 완료!")

🎯 변형문제 2 - 텍스트 데이터 파일 다운로드해서 저장하기

인터넷 주소에 있는 텍스트(csv) 파일을 다운로드해서 weather_data.csv라는 이름으로 내 컴퓨터에 저장하는 코드를 작성해보세요. (이미지와 달리 텍스트 파일이므로 인코딩 처리가 필요합니다)

정답 보기 
import requests

data_url = "http://example.com/data/weather.csv"

print("데이터 파일을 요청하는 중...")
response = requests.get(data_url)
response.encoding = 'utf-8'

with open('weather_data.csv', 'w', encoding='utf-8') as f:
    f.write(response.text)

print("CSV 파일 저장 완료!")
# 핵심 차이: 이미지는 바이너리(wb + response.content),
# 텍스트 파일은 텍스트 모드(w + response.text)로 저장
