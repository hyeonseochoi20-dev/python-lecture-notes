웹 크롤링이란?

웹사이트로부터 데이터를 자동으로 수집하는 기술입니다.

주요 단계:

  1. 요청 보내기 (Request)
  2. 응답 받기 (Response)
  3. 응답 파싱 (Parsing)
  4. 필요한 데이터 추출 (Extraction)

에러 처리 및 타임아웃

import requests
from requests.exceptions import *

try:
    response = requests.get(
        'https://httpbin.org/delay/5',  # 5초 지연
        timeout=3  # 3초 타임아웃
    )
except Timeout:
    print("요청이 시간 초과되었습니다")
except ConnectionError:
    print("네트워크 연결 오류")
except HTTPError as e:
    print(f"HTTP 오류: {e.response.status_code}")
except RequestException as e:
    print(f"요청 중 오류 발생: {e}")
else:
    print(f"성공! 상태 코드: {response.status_code}")
finally:
    print("요청 처리 완료")

예시 1: GitHub 사용자 정보 가져오기

import requests

def get_github_user(username):
    """GitHub 사용자 정보 가져오기"""
    url = f"https://api.github.com/users/{username}"
    headers = {
        'Accept': 'application/vnd.github.v3+json',
        'User-Agent': 'Python-Crawling-Tutorial/1.0'
    }

    try:
        response = requests.get(url, headers=headers, timeout=10)
        response.raise_for_status()  # 4xx/5xx 오류 시 예외 발생

        user_data = response.json()
        return {
            'login': user_data['login'],
            'name': user_data.get('name'),
            'public_repos': user_data['public_repos'],
            'followers': user_data['followers'],
            'html_url': user_data['html_url']
        }
    except requests.exceptions.HTTPError as e:
        if response.status_code == 404:
            print(f"사용자 '{username}'을 찾을 수 없습니다")
        else:
            print(f"HTTP 오류: {response.status_code}")
        return None
    except requests.exceptions.RequestException as e:
        print(f"요청 오류: {e}")
        return None

# 사용

    print(f"팔로워: {user_info['followers']}명")

import requests

def get_weather(city_name, api_key):
    """특정 도시의 날씨 정보 가져오기"""
    base_url = "http://api.openweathermap.org/data/2.5/weather"
    params = {
        'q': city_name,
        'appid': api_key,
        'units': 'metric',  # 섭씨 온도
        'lang': 'kr'        # 한국어 설명
    }

    try:
        response = requests.get(base_url, params=params, timeout=10)
        response.raise_for_status()

        data = response.json()
        return {
            'city': data['name'],
            'country': data['sys']['country'],
            'temp': data['main']['temp'],
            'feels_like': data['main']['feels_like'],
            'humidity': data['main']['humidity'],
            'description': data['weather'][0]['description'],
            'wind_speed': data['wind']['speed']
        }
    except requests.exceptions.HTTPError as e:
        if response.status_code == 401:
            print("API 키가 유효하지 않습니다")
        elif response.status_code == 404:
            print(f"도시 '{city_name}'을 찾을 수 없습니다")
        else:
            print(f"HTTP 오류: {response.status_code}")
        return None
    except requests.exceptions.RequestException as e:
        print(f"요청 오류: {e}")
        return None

재시도 메커니즘 (Retry Strategy)

import requests
from requests.adapters import HTTPAdapter
from urllib3.util.retry import Retry

def requests_retry_session(
    retries=3,
    backoff_factor=0.3,
    status_forcelist=(500, 502, 504),
    session=None,
):
    session = session or requests.Session()
    retry = Retry(
        total=retries,
        read=retries,
        connect=retries,
        backoff_factor=backoff_factor,
        status_forcelist=status_forcelist,
    )
    adapter = HTTPAdapter(max_retries=retry)
    session.mount('http://', adapter)
    session.mount('https://', adapter)
    return session

# 사용
session = requests_retry_session(retries=3, backoff_factor=1)
try:
    response = session.get('https://httpbin.org/status/500', timeout=5)
    response.raise_for_status()
except requests.exceptions.RequestException as e:
    print(f"최종적으로 실패: {e}")
finally:
    session.close()

HTTP 메소드 (서버와의 통신 방식)

HTTP 메소드는 총 9가지가 있지만, 실무 크롤링/API 통신에서는 GET과 POST 두 가지를 압도적으로 많이 사용합니다.

메소드용도
GET데이터 조회 (요청 데이터를 URL의 ? 뒤에 담아 전달)
POST데이터 생성/전송 (요청 데이터를 HTTP Body에 담아 전달, 로그인/파일업로드 등)
PUT / PATCH데이터 수정 (PUT은 전체 교체, PATCH는 부분 수정)
DELETE데이터 삭제
HEAD / OPTIONS / CONNECT / TRACE메타정보 조회, 통신 옵션 확인 등 (거의 사용 안 함)
import requests

# GET: 파라미터를 URL 뒤에 ?key=value 형태로 붙여서 요청
res = requests.get(url, params={'query': '검색어'})

# POST: 데이터를 HTTP Body에 담아서 요청
res = requests.post(url, data={'id': 'user1', 'pw': '1234'})
# 또는 JSON 형식으로
res = requests.post(url, json={'id': 'user1', 'pw': '1234'})

예시 2: 요청 헤더를 활용해 접근 차단 우회하기

일부 API는 정상적인 브라우저 요청처럼 보이지 않으면 응답을 거부합니다. 이럴 때는 개발자도구의 Network 탭에서 실제 브라우저가 보내는 요청 헤더(Request Headers)를 확인하고, 필요한 헤더 값을 그대로 requests에 추가하면 해결되는 경우가 많습니다.

import requests

url = 'https://finance.daum.net/api/exchanges/summaries'
req_headers = {
    'user-agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36',
    'referer': 'https://finance.daum.net/exchanges',
}
res = requests.get(url, headers=req_headers)
print(res.status_code)
print(res.json())

핵심 포인트: API가 401/403 등의 오류를 반환한다면 가장 먼저 User-Agent와 Referer 헤더부터 브라우저와 동일하게 맞춰보는 것이 정석적인 접근법입니다.

import requests
from datetime import datetime

def get_korean_holidays(year):
    """한국 공공데이터 포털의 공휴일 API 사용 (예시)"""
    # 실제로는 서비스 키가 필요합니다
    url = "http://apis.data.go.kr/B090041/openapi/service/SpcdeInfoService/getRestDeInfo"
    params = {
        'ServiceKey': 'YOUR_SERVICE_KEY',  # 실제 서비스 키 필요
        'solYear': year,
        '_type': 'json'  # JSON 응답 요청
    }

    try:
        response = requests.get(url, params=params, timeout=10)
        response.raise_for_status()

        data = response.json()
        items = data['response']['body']['items']['item']

        holidays = []
        if isinstance(items, list):  # 여러 개인 경우
            for item in items:
                holidays.append({
                    'date': f"{item['locdate'][:4]}-{item['locdate'][4:6]}-{item['locdate'][6:]}",
                    'name': item['dateName'],
                    'isHoliday': item['isHoliday'] == 'Y'
                })
        elif isinstance(items, dict):  # 단일 항목인 경우
            holidays.append({
                'date': f"{items['locdate'][:4]}-{items['locdate'][4:6]}-{items['locdate'][6:]}",
                'name': items['dateName'],
                'isHoliday': items['isHoliday'] == 'Y'
            })

        return holidays
    except Exception as e:
        print(f"공휴일 정보 가져오기 실패: {e}")

import requests
import os
from urllib.parse import urljoin, urlparse

def download_image(url, folder="downloads"):
    """이미지 URL로부터 이미지 다운로드"""
    try:
        # 폴더 생성
        os.makedirs(folder, exist_ok=True)

        # 헤더 정보만 먼저 가져오기 (효율성)
        head_response = requests.head(url, timeout=10)
        head_response.raise_for_status()

        # 콘텐츠 타입 확인
        content_type = head_response.headers.get('content-type', '').lower()
        if not content_type.startswith('image/'):
            print(f"경고: {url}은 이미지가 아닌 것 같습니다 (Content-Type: {content_type})")
            # 그래도 시도해볼 수 있음

        # 실제 다운로드
        response = requests.get(url, stream=True, timeout=30)
        response.raise_for_status()

        # 파일명 추출
        parsed_url = urlparse(url)
        filename = os.path.basename(parsed_url.path)
        if not filename or '.' not in filename:
            # URL에서 파일명을 추출할 수 없는 경우
            content_type = response.headers.get('content-type', '')
            if 'jpeg' in content_type or 'jpg' in content_type:
                filename = "image.jpg"
            elif 'png' in content_type:
                filename = "image.png"
            elif 'gif' in content_type:
                filename = "image.gif"
            else:
                filename = "download.bin"

        filepath = os.path.join(folder, filename)

        # 청크 단위로 다운로드 (대용량 파일 처리)
        total_size = int(response.headers.get('content-length', 0))
        downloaded = 0

        with open(filepath, 'wb') as f:
            for chunk in response.iter_content(chunk_size=8192):
                if chunk:  # 청크가 비어있지 않은 경우
                    f.write(chunk)
                    downloaded += len(chunk)
                    # 진행률 표시 (선택사항)
                    if total_size > 0:
                        progress = (downloaded / total_size) * 100
                        print(f"\r다운로드 진행: {progress:.1f}%", end="", flush=True)

        print(f"\n다운로드 완료: {filepath}")
        return filepath

    except requests.exceptions.RequestException as e:
        print(f"다운로드 실패: {e}")
        return None
